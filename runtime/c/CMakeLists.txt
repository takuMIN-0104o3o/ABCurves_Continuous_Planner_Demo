cmake_minimum_required(VERSION 3.16)
project(abcurves_renderer C)
add_library(abcurves_renderer SHARED src/abc_online.c src/abc_fixed.c src/abc_w5_stream.c)
target_include_directories(abcurves_renderer PUBLIC include PRIVATE src)
target_compile_definitions(abcurves_renderer PRIVATE ABC_RENDERER_EXPORTS)
set_target_properties(abcurves_renderer PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO C_VISIBILITY_PRESET hidden)
if(MSVC)
  target_compile_options(abcurves_renderer PRIVATE /W4 /fp:strict)
  target_link_options(abcurves_renderer PRIVATE /Brepro /INCREMENTAL:NO)
else()
  target_compile_options(abcurves_renderer PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(abcurves_renderer PRIVATE m)
endif()
